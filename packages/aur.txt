# AUR packages, installed by bootstrap.sh via paru.
# One per line. Blank lines and # comments are ignored — same parser as
# pacman.txt (see install_list() in bootstrap.sh).

# --- Phase 3: dev toolchain ---------------------------------------------------

# Temporal workflow engine CLI. Not in the official repos.
temporal-cli

# API client (Postman/Insomnia alternative), Electron-based. -bin variant:
# prebuilt binary, avoids an Electron/Node build from source at install
# time (same reasoning as paru-bin, minus the libalpm ABI risk that made
# us avoid paru-bin specifically — see CLAUDE.md gotchas).
bruno-bin

# --- Phase 5: Hyprland ---------------------------------------------------------

# Bitmap pixel font, used for notification titles/timestamps in swaync's
# F1 theme (themes/f1/swaync.css) — not in the official repos.
ttf-press-start-2p

# TokyoNight-colored GTK3/4 widget theme (buttons, scrollbars,
# selection) — packages the well-established Fausto-Korpsvart/
# Tokyo-Night-GTK-Theme project. Installs as "Tokyonight-Dark"
# (confirmed via `ls /usr/share/themes/` — not assumed from the
# package name), which is the actual value themes/tokyonight/meta.sh
# passes to gsettings.
tokyonight-gtk-theme-git

# Cursor theme. -bin variant (see bruno-bin above for the same
# reasoning) — cursor themes are just data assets, no reason to build
# from source. Bibata-Modern-Ice specifically (of several variants this
# package installs) for its cool blue tint, closer to the TokyoNight
# accent than Amber/Classic.
bibata-cursor-theme-bin

# --- Phase 5: quick-access keybinds ---------------------------------------

# Power menu (lock/logout/suspend/reboot/shutdown grid), bound to
# $mod+P. Not in the official repos (confirmed via `pacman -Si wlogout`
# — AUR only). Styled per-theme like waybar/rofi/swaync — see
# wlogout/.config/wlogout/ and themes/tokyonight/wlogout.css.
wlogout
