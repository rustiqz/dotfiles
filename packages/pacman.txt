# Official-repo packages, installed by bootstrap.sh via pacman.
# One per line. Blank lines and # comments are ignored.
# Grouped by phase so this stays readable as it grows.

# --- Phase 1: base system ----------------------------------------------------

# Pacman tooling
pacman-contrib          # paccache (cache cleanup), checkupdates, rankmirrors
reflector               # regenerates the mirrorlist ranked by speed
pkgfile                 # "which package owns this command?" lookups

# Audio: PipeWire replaces PulseAudio + JACK behind compatible interfaces,
# so apps expecting either still work. wireplumber is its session manager —
# it decides routing and device priorities. Without it PipeWire runs but
# nothing is wired up.
pipewire
pipewire-alsa           # ALSA apps route through PipeWire
pipewire-pulse          # PulseAudio drop-in replacement
pipewire-jack           # JACK drop-in replacement
wireplumber
pavucontrol             # GUI mixer, useful for per-app volume debugging

# Bluetooth stack. bluez is the daemon, bluez-utils gives you bluetoothctl
# for CLI pairing — worth having since Hyprland has no bluetooth applet
# until you configure one.
bluez
bluez-utils

# Networking (GNOME already pulled NetworkManager in; listed so a fresh
# machine without GNOME still gets it)
networkmanager
openssh

# Memory: compressed swap in RAM. Faster than disk swap and avoids
# writing swap to your SSD.
zram-generator

# Firmware updates over LVFS — mostly relevant for laptops
fwupd

# Fonts. A Nerd Font is non-optional here: starship, eza icons, lazygit
# and most Neovim statuslines render glyphs from that patched range.
# Without it you get boxes everywhere.
ttf-jetbrains-mono-nerd
noto-fonts              # broad Unicode coverage, the sane default
noto-fonts-emoji        # colour emoji
noto-fonts-cjk          # Chinese/Japanese/Korean, prevents tofu in browsers
ttf-liberation          # metric-compatible with Arial/Times, fixes web pages

# Misc base
man-db
man-pages
xdg-user-dirs           # creates ~/Documents, ~/Downloads etc.
unzip
zip
rsync


# --- Phase 1 additions -------------------------------------------------------

# RealtimeKit: grants realtime scheduling priority to audio threads on
# request. PipeWire logs "RTKit does not give us MaxRealtimePriority"
# and falls back to priority 1 without it, which risks xruns/crackling
# when the CPU is busy. Optional dep of pipewire, so not pulled in.
rtkit

# glxinfo / eglinfo — inspect which Mesa driver and GL version are active.
# Diagnostic only, but the first thing you want when a compositor
# misbehaves.
mesa-utils


# --- Phase 2: shell & terminal -----------------------------------------------

# Terminal emulator. GPU-accelerated, config is a single plain-text file —
# no GUI settings to lose track of.
kitty

# Shell. Login shell is switched to this in hooks/40-shell.sh; the config
# itself lives in the zsh/ stow package.
zsh
zsh-autosuggestions        # ghost-text completion from history as you type
zsh-syntax-highlighting    # must be sourced last in .zshrc — see zsh/.zshrc
zsh-history-substring-search  # up/down arrow filters history by what's typed

# Prompt. Single static binary, config is TOML — no shell-specific prompt
# logic to maintain per-shell if the login shell ever changes.
starship

# Core CLI replacements/additions, all referenced from zsh/.zshrc aliases
# or shell init:
fzf                        # fuzzy finder — powers Ctrl-R history search
ripgrep                    # rg: grep replacement, respects .gitignore
fd                         # find replacement, sane defaults + .gitignore
bat                        # cat replacement: syntax highlighting, git gutter
eza                        # ls replacement: icons, git status, tree view
zoxide                     # cd replacement: jumps to frecent directories
git-delta                  # git diff/log pager, configured in git's stow pkg
btop                       # process/resource monitor, mouse + theme support
lazygit                    # terminal git UI, config themed to match the rest
jq                         # JSON query/pretty-print, used constantly at a CLI

# Not otherwise guaranteed on a minimal Arch install; hooks/41-bat-theme.sh
# needs it to fetch the TokyoNight theme.
curl


# --- Phase 3: dev toolchain ---------------------------------------------------

# Node version manager. Manages Node itself (see mise/.config/mise/config.toml
# for the pinned default version) — deliberately NOT installing `nodejs`
# here, since that would fight mise for control of which Node is on PATH.
mise

# Node package manager. Only depends on node-gyp (for native addon builds),
# not on `nodejs` itself — it bundles its own runtime, so it coexists with
# mise-managed Node versions without conflict.
pnpm

# Containers. docker-compose here is the v2 CLI plugin (`docker compose`),
# not the old standalone python `docker-compose` binary.
docker
docker-compose

# Postgres client. Arch ships psql bundled with the full postgresql package
# (server binaries included) — there's no client-only split like Debian's
# postgresql-client. The server is never enabled; Postgres itself is
# expected to run in Docker, this is just the CLI to talk to it.
postgresql

# Redis-protocol client. Arch dropped the official `redis` package after
# Redis's 2024 relicensing; valkey is the actively-maintained fork Arch
# ships instead. Same story as postgresql above — server unused, this is
# just for `valkey-cli` to talk to a container.
valkey

# Per-directory environment variables (.envrc), auto-loaded/unloaded on cd.
direnv


# --- Phase 4: editor -----------------------------------------------------------

# Editor itself. LSP servers (vtsls, lua_ls, eslint) are NOT system
# packages — mason.nvim installs those into ~/.local/share/nvim/mason on
# first launch. See nvim/.config/nvim/lua/plugins/lsp.lua.
neovim

# Terminal multiplexer. Plugins (tmux-resurrect/continuum) are bootstrapped
# by hooks/60-tmux-plugins.sh via TPM, not this manifest — TPM itself is
# git-cloned there since it isn't a pacman package.
tmux

# Wayland clipboard tool. tmux's copy-mode `y` binding pipes selections
# through `wl-copy` (tmux/.tmux.conf) rather than tmux's own paste buffer,
# so yanked text is available to every other app too.
wl-clipboard

# --- Phase 5: Hyprland ---------------------------------------------------------

# The compositor itself. Installing this package alone is what makes a
# "Hyprland" session appear on the GDM login screen — no separate login
# manager needed, GNOME's GDM enumerates any *.desktop file under
# /usr/share/wayland-sessions.
hyprland

# Lets Wayland apps request things the compositor itself doesn't implement:
# screen share/recording (browsers, Discord), the GTK/Qt "open file" dialog,
# etc. Without it those either hang or silently fail. Hyprland-specific
# because the generic xdg-desktop-portal-wlr doesn't know Hyprland's
# IPC protocol for window/output listing.
xdg-desktop-portal-hyprland

# Qt apps render through XWayland (blurry, no native Wayland input) without
# these. Most of this repo's apps are GTK or Electron, but installing both
# now avoids a surprise later when some Qt-based tool gets added.
qt5-wayland
qt6-wayland

# Status bar: workspaces, clock, network, volume, battery, power profile.
# Hyprland draws windows only — no bar, tray, or clock without this.
# Native Wayland layer-shell client, which is why it docks properly
# instead of floating like an X11 panel would.
waybar

# App launcher (replaces GNOME's Activities overview). Plain `rofi`, not
# the old `rofi-wayland` AUR fork — as of rofi 2.0.0 the Wayland backend
# was merged into upstream itself, and this package `Provides`/`Replaces`
# rofi-wayland outright (confirmed via `pacman -Si rofi`; the AUR fork no
# longer exists). No AUR package needed for this at all.
rofi

# GTK Bluetooth manager (pairing, trust, device list) — the GUI behind
# waybar's bluetooth module click, same role pavucontrol plays for the
# pulseaudio module. bluez/bluez-utils (bluetoothctl, the daemon itself)
# already installed in Phase 1; this is just the front-end.
blueman

# nm-connection-editor — the GUI behind waybar's network module dropdown
# ("Open Network Settings…"). Was already present on this machine (pulled
# in some other way, likely as a GNOME dependency) but was never actually
# tracked here — a fresh bootstrap elsewhere would have silently missed
# it and that menu item would just fail.
network-manager-applet

# Notification daemon + panel (config: swaync/.config/swaync/). Conflicts
# on the org.freedesktop.Notifications D-Bus name with any other
# notification daemon — this machine had a stray, untracked dunst.service
# claiming it (masked in hooks/70-swaync.sh) before swaync could start.
swaync

# Clipboard history for Wayland — GNOME's clipboard history lives in a
# Shell extension, which has no equivalent under Hyprland. Stores
# whatever wl-paste --watch feeds it (hypr/autostart.conf); picking an
# entry goes through rofi (already installed) via a plain pipe, per
# cliphist's own "no concept of a picker, only pipes" design.
cliphist

# Screen lock (hyprlock), idle daemon (hypridle), and wallpaper daemon
# (hyprpaper) — all three from the Hyprland team specifically, same
# config-fragment convention as the compositor itself. hypridle's
# lock_cmd is hyprlock; hyprlock's background and hyprpaper's wallpaper
# both read through the same ~/.local/share/wallpapers/.current symlink
# (hooks/71-wallpaper.sh), so picking a new wallpaper (~/bin/wallpaper)
# updates both together.
hyprlock
hypridle
hyprpaper

# Screenshots. grim captures a Wayland output/region to a PNG; slurp
# drives the interactive drag-to-select box grim's -g geometry flag
# consumes; swappy is the annotate/crop/arrow editor that grim's output
# gets piped into before saving (bin/bin/screenshot's --edit path).
# grim and slurp were already present on this machine (pulled in some
# other way, likely a GNOME dependency) but untracked here; swappy is
# genuinely new.
grim
slurp
swappy

# Polkit authentication agent — GNOME's own is tied to gnome-shell, so
# nothing puts a password dialog on screen for privileged D-Bus actions
# under Hyprland without this running (checked this machine's actual
# /usr/share/polkit-1/{actions,rules.d}/ to see what that covers in
# practice — most of it, like NetworkManager system-connection edits,
# is auto-granted to wheel-group users by local rules and never
# prompts at all; timedate1/hostname1 changes are the realistic case).
# Hyprland team's own agent (Qt/QML), chosen for consistency with
# hyprlock/hypridle/hyprpaper over GTK (polkit-gnome, unmaintained
# upstream) or KDE/LXQt, which drag in much heavier dependency chains.
hyprpolkitagent

# Tried as the fix for hyprpolkitagent's "QQuickStyle::setStyle() must
# be called before loading QML" warning (hyprwm/hyprpolkitagent#10),
# based on that issue's discussion. Turned out to likely be a red
# herring on this version: hyprpolkitagent actually requests its own
# style, org.hyprland.style, which already ships on disk via
# hyprland-qt-support (a hard dependency, confirmed via `pacman -Ql`)
# — installing this didn't change the dialog's plain appearance.
# Harmless to keep installed; the actual fix was the Qt theming setup
# below (qt6ct + QT_QPA_PLATFORMTHEME) — confirmed working once that
# was in place.
qqc2-desktop-style

# --- Phase 5: whole-desktop theme switcher --------------------------------

# TokyoNight-consistent icon set. Folder-color tinting via
# papirus-folders wasn't set up — the default blue already reads close
# enough to the TokyoNight accent without adding another moving part.
papirus-icon-theme

# Qt5/Qt6 settings apps + the platform-theme bridge that actually makes
# Qt apps read them (QT_QPA_PLATFORMTHEME=qt6ct, themes/tokyonight/
# uwsm-env.sh) — without this, Qt apps (hyprpolkitagent included) ignore
# qt6ct/qt5ct entirely regardless of how their .conf files are set up.
# Both own and rewrite their own config file on every launch (confirmed
# by testing — a hand-written comment survived exactly one launch
# before being mangled), which is why qt6ct.conf/qt5ct.conf/
# themes/tokyonight/qt-colors.conf carry no comments of their own; the
# reasoning lives here and in the commit message instead.
qt6ct
qt5ct

# --- Phase 5: quick-access keybinds ---------------------------------------

# Emoji/unicode/character picker, bound to $mod+period (matches the
# Windows-key-plus-period muscle memory this shortcut is named after).
# Drives its picker window through the already-installed `rofi` binary,
# so it automatically inherits rofi/.config/rofi/active-theme.rasi with
# no styling of its own needed.
rofimoji

# Browser, bound to $mod+B. Was already installed outside this repo
# (like dolphin/network-manager-applet before it) — tracked now since
# the keybind depends on it; a clean-VM bootstrap would otherwise not
# have it. Store-installed themes (Nighttime, Scuderia Ferrari F1) live
# in the profile itself, not this repo — see CLAUDE.md's Current state
# for why that's left alone rather than reproduced here.
firefox

# --- Phase 5: wallpaper-driven theme (matugen) ------------------------------

# Material You palette generator: extracts a seed color from the
# current wallpaper and fills in matugen/templates/*.tpl to produce
# themes/wallpaper/* — a theme like tokyonight/f1, except generated
# instead of hand-written. Triggered by `bin/bin/theme wallpaper`
# (matugen/config.toml), never run automatically on a wallpaper pick —
# see CLAUDE.md for why that stayed a deliberate separate step.
matugen
